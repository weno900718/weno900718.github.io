const getLunarDate = (date) => {

  try {

    const formatter = new Intl.DateTimeFormat('zh-TW-u-ca-chinese', { day: 'numeric', month: 'numeric' });

    const parts = formatter.formatToParts(date);

    const dVal = parts.find(p => p.type === 'day')?.value;

    const mVal = parts.find(p => p.type === 'month')?.value;

    if (!dVal || !mVal) return '';

    const numD = parseInt(dVal); const numM = parseInt(mVal);

    const ms = ['','正月','二月','三月','四月','五月','六月','七月','八月','九月','十月','冬月','臘月'];

    const ds = ['','一','二','三','四','五','六','七','八','九','十'];

    let lunar = '';

    if (numD <= 10) lunar = `初${ds[numD]}`;

    else if (numD < 20) lunar = `十${ds[numD % 10]}`;

    else if (numD === 20) lunar = '二十';

    else if (numD < 30) lunar = `廿${ds[numD % 10]}`;

    else if (numD === 30) lunar = '三十';

    return numD === 1 ? ms[numM] : lunar;

  } catch (e) { return ''; }

};

// --- Sub-Component: Drawing Canvas ---

const DrawingPad = ({ onSave, onCancel }) => {

  const canvasRef = useRef(null);

  const [isDrawing, setIsDrawing] = useState(false);

  const [ctx, setCtx] = useState(null);

  useEffect(() => {

    const canvas = canvasRef.current;

    if (!canvas) return;

    canvas.width = canvas.parentElement.offsetWidth; canvas.height = 320;

    const context = canvas.getContext('2d');

    context.lineCap = 'round'; context.strokeStyle = '#4f46e5'; context.lineWidth = 4;

    context.fillStyle = '#ffffff'; context.fillRect(0, 0, canvas.width, canvas.height);

    setCtx(context);

  }, []);

  const getPos = (e) => {

    const rect = canvasRef.current.getBoundingClientRect();

    const clientX = e.touches ? e.touches[0].clientX : e.clientX;

    const clientY = e.touches ? e.touches[0].clientY : e.clientY;

    return { x: clientX - rect.left, y: clientY - rect.top };

  };

  return (

    <div className="flex flex-col space-y-4">

      <div className="border-2 border-dashed border-indigo-200 rounded-2xl overflow-hidden touch-none bg-white shadow-inner">

        <canvas ref={canvasRef}

          onMouseDown={(e) => { setIsDrawing(true); ctx.beginPath(); const p = getPos(e); ctx.moveTo(p.x, p.y); }}

          onMouseMove={(e) => { if (!isDrawing) return; const p = getPos(e); ctx.lineTo(p.x, p.y); ctx.stroke(); }}

          onMouseUp={() => setIsDrawing(false)}

          onTouchStart={(e) => { e.preventDefault(); setIsDrawing(true); ctx.beginPath(); const p = getPos(e); ctx.moveTo(p.x, p.y); }}

          onTouchMove={(e) => { e.preventDefault(); if (!isDrawing) return; const p = getPos(e); ctx.lineTo(p.x, p.y); ctx.stroke(); }}

          onTouchEnd={() => setIsDrawing(false)}

          className="w-full h-[320px] cursor-crosshair"

        />

      </div>

      <div className="flex justify-between px-1">

        <button onClick={() => { ctx.fillStyle = '#ffffff'; ctx.fillRect(0, 0, canvasRef.current.width, canvasRef.current.height); }} className="text-red-500 font-bold text-xs bg-red-50 px-4 py-2 rounded-xl active:scale-95 transition-all">清除重畫</button>

        <div className="space-x-3">

          <button onClick={onCancel} className="text-gray-400 font-bold text-xs px-4 py-2 active:scale-95">取消</button>

          <button onClick={() => onSave(canvasRef.current.toDataURL('image/png'))} className="bg-indigo-600 text-white px-6 py-2 rounded-xl font-bold text-xs shadow-lg active:scale-95">完成手繪</button>

        </div>

      </div>

    </div>

  );

};

            <div className="flex flex-col gap-1.5"><label className="text-[10px] font-black text-slate-400 uppercase pl-1 tracking-widest">內容詳情</label><textarea rows="8" value={noteForm.content} onChange={e=>setNoteForm({...noteForm, content:e.target.value})} className="w-full p-5 bg-slate-50 rounded-3xl text-xs outline-none focus:ring-2 focus:ring-indigo-200 font-medium leading-relaxed" placeholder="分享此刻的心情與觀察..." /></div>

            <div className="flex flex-col gap-1.5"><label className="text-[10px] font-black text-slate-400 uppercase pl-1 tracking-widest">標籤分類</label><input type="text" value={noteForm.tags} onChange={e=>setNoteForm({...noteForm, tags:e.target.value})} className="w-full p-4 bg-slate-100 rounded-2xl outline-none font-bold text-xs uppercase tracking-widest" placeholder="例如：美食, 旅遊 (逗號分隔)" /></div>

            <button onClick={handleSaveNote} className="w-full bg-indigo-600 text-white p-5 rounded-3xl font-black shadow-2xl active:scale-95 transition-all text-sm uppercase tracking-widest">{editingId?'確認更新記事紀錄':'確認儲存紀錄'}</button>

          </div>

        ) : (

          <DrawingPad onSave={(d)=>{setNoteForm(p=>({...p, attachments:[...p.attachments, {url:d}]})); setIsDrawingMode(false);}} onCancel={()=>setIsDrawingMode(false)} />

        )}

      </Modal>

      {/* Accounting Modal */}

      <Modal isOpen={showTransModal} onClose={()=>{setShowTransModal(false); setEditingId(null);}} title={editingId?'編輯明細':'快速記帳中心'}>

        <div className="space-y-6 text-center animate-fade-in font-bold text-slate-700">

          <div className="flex bg-slate-100 p-1.5 rounded-2xl">

            <button onClick={()=>setTransForm({...transForm, type:'expense'})} className={`flex-1 py-3 rounded-xl text-xs font-black transition-all ${transForm.type==='expense'?'bg-white text-rose-600 shadow-sm' : 'text-gray-400'}`}>支出項目</button>

            <button onClick={()=>setTransForm({...transForm, type:'income'})} className={`flex-1 py-3 rounded-xl text-xs font-black transition-all ${transForm.type==='income'?'bg-white text-emerald-600 shadow-sm' : 'text-gray-400'}`}>收入項目</button>

          </div>

          <input type="number" value={transForm.amount} onChange={e=>setTransForm({...transForm, amount:e.target.value})} className="w-full text-3xl font-black text-center focus:outline-none bg-transparent text-indigo-600 tracking-tighter" placeholder="0" />

          <div className="space-y-4 text-left text-sm font-bold">

            <div className="flex flex-col gap-1.5"><label className="text-[10px] font-black text-slate-400 uppercase pl-1 tracking-widest">交易日期</label><input type="date" value={transForm.date} onChange={e=>setTransForm({...transForm, date:e.target.value})} className="w-full p-4 bg-slate-50 rounded-2xl outline-none font-black" /></div>

            <div className="flex flex-col gap-1.5"><label className="text-[10px] font-black text-slate-400 uppercase pl-1 tracking-widest">分類選擇</label><select value={transForm.category} onChange={e=>setTransForm({...transForm, category:e.target.value})} className="w-full p-4 bg-slate-50 rounded-2xl outline-none font-black bg-white border-none">{transForm.type==='expense'?['飲食','交通','購物','居家','娛樂','旅費','其他'].map(c=><option key={c} value={c}>{c}</option>):['薪資','獎金','投資','其他'].map(c=><option key={c} value={c}>{c}</option>)}</select></div>

            <div className="flex flex-col gap-1.5"><label className="text-[10px] font-black text-slate-400 uppercase pl-1 tracking-widest">簡短描述</label><input type="text" value={transForm.description} onChange={e=>setTransForm({...transForm, description:e.target.value})} className="w-full p-4 bg-slate-50 rounded-2xl outline-none border-2 border-transparent focus:border-indigo-400 focus:bg-white transition-all font-black" placeholder="備註內容..." /></div>

          </div>

          <button onClick={handleSaveTrans} className="w-full bg-emerald-600 text-white p-5 rounded-3xl font-black shadow-2xl active:scale-95 transition-all text-sm uppercase tracking-widest tracking-tighter">確認儲存數據</button>

        </div>

      </Modal>

      {/* Plan Modal (優化：日期左右並排) */}

      <Modal isOpen={showPlanModal} onClose={()=>{setShowPlanModal(false); setEditingId(null);}} title={editingId?'編輯旅程':'規劃新旅遊'} isSaving={isSaving}>

        <div className="space-y-4 animate-fade-in font-bold text-slate-700 text-left text-sm">

          <div className="flex flex-col gap-1.5"><label className="text-[10px] font-black text-slate-400 uppercase pl-1 tracking-widest">行程計畫主題</label><input type="text" value={planForm.title} onChange={e=>setPlanForm({...planForm, title:e.target.value})} className="w-full p-4 bg-slate-50 rounded-2xl outline-none border-2 border-transparent focus:border-indigo-400 focus:bg-white transition-all font-black" placeholder="如: 日本五天四夜之旅" /></div>

         

          {/* 日期左右並排佈局 */}

          <div className="grid grid-cols-2 gap-3">

            <div className="flex flex-col gap-1.5">

              <label className="text-[9px] font-black text-slate-300 block pl-1 uppercase tracking-widest">開始日期</label>

              <input type="date" value={planForm.startDate} onChange={e=>setPlanForm({...planForm, startDate:e.target.value})} className="w-full p-4 bg-slate-50 rounded-2xl outline-none font-black shadow-sm text-xs"/>

            </div>

            <div className="flex flex-col gap-1.5">

              <label className="text-[9px] font-black text-slate-300 block pl-1 uppercase tracking-widest">結束日期</label>

              <input type="date" value={planForm.endDate} onChange={e=>setPlanForm({...planForm, endDate:e.target.value})} className="w-full p-4 bg-slate-50 rounded-2xl outline-none font-black shadow-sm text-xs"/>

            </div>

          </div>

          <div className="flex flex-col gap-1.5"><label className="text-[10px] font-black text-slate-400 uppercase pl-1 tracking-widest">主要目的地點</label><input type="text" value={planForm.location} onChange={e=>setPlanForm({...planForm, location:e.target.value})} className="w-full p-4 bg-slate-50 rounded-2xl outline-none font-black shadow-sm" placeholder="目的地 (如: 東京, 那霸)" /></div>

          <div className="flex flex-col gap-1.5"><label className="text-[10px] font-black text-slate-400 uppercase pl-1 tracking-widest">詳細規劃說明</label><textarea rows="4" value={planForm.description} onChange={e=>setPlanForm({...planForm, description:e.target.value})} className="w-full p-5 bg-slate-50 rounded-3xl text-xs outline-none focus:ring-2 focus:ring-orange-200 font-medium leading-relaxed shadow-sm" placeholder="航班、飯店、必訪景點規劃細節..." /></div>

          <button onClick={handleSavePlan} className="w-full bg-orange-500 text-white p-5 rounded-3xl font-black shadow-2xl active:scale-95 transition-all text-sm uppercase tracking-widest tracking-tighter">確認儲存旅遊計畫</button>

        </div>

      </Modal>

    </div>

  );

}
